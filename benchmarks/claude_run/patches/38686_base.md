```python
_BUILTIN_TOOL_PREFIXES = (
    "advisor_",
    "bash_",
    "code_execution_",
    "computer_",
    "memory_",
    "text_editor_",
    "web_fetch_",
    "web_search_",
)


def _is_builtin_tool(tool: Any) -> bool:
    """Check whether a tool is an Anthropic provider-native (server-side) tool."""
    if not isinstance(tool, dict):
        return False
    tool_type = tool.get("type")
    return isinstance(tool_type, str) and tool_type.startswith(_BUILTIN_TOOL_PREFIXES)
```

```python
def test_bind_tools_advisor_builtin_tool() -> None:
    """Advisor built-in tools are passed through unchanged (see #38644)."""
    llm = ChatAnthropic(model="claude-sonnet-4-5-20250929")  # type: ignore[call-arg]

    advisor_tool = {
        "type": "advisor_20260301",
        "name": "advisor",
    }

    chain = llm.bind_tools([advisor_tool])
    assert chain.kwargs["tools"] == [advisor_tool]  # type: ignore[attr-defined]


def test_is_builtin_tool_advisor_prefix() -> None:
    assert _is_builtin_tool({"type": "advisor_20260301", "name": "advisor"})
    assert not _is_builtin_tool({"type": "custom_tool", "name": "custom"})
```
