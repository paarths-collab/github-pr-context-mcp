_BUILTIN_TOOL_PREFIXES = (
    "advisor_",
    "bash_",
    "code_execution_",
    "computer_",
    "memory_",
    "text_editor_",
    "web_fetch_",
    "web_search_",
)


def _is_builtin_tool(tool: Any) -> bool:
    if not isinstance(tool, dict):
        return False
    tool_type = tool.get("type")
    return isinstance(tool_type, str) and tool_type.startswith(_BUILTIN_TOOL_PREFIXES)
```

def test_bind_tools_builtin_advisor_tool() -> None:
    """Advisor built-in tools are passed through unchanged (no KeyError)."""
    llm = ChatAnthropic(model=MODEL_NAME)  # type: ignore[call-arg]
    advisor_tool = {"type": "advisor_20260301", "name": "advisor"}

    assert _is_builtin_tool(advisor_tool)

    bound = llm.bind_tools([advisor_tool])
    assert bound.kwargs["tools"] == [advisor_tool]  # type: ignore[attr-defined]
```
