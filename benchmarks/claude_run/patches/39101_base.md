def _format_text_block(block: dict) -> dict:
    """Narrow a text content block to the fields the Anthropic API accepts.

    LangChain v1 content blocks may carry extra bookkeeping fields (e.g. ``id``)
    that Anthropic rejects with
    ``400 ...: Extra inputs are not permitted``. Only ``type``, ``text``,
    ``cache_control`` and ``citations`` are forwarded, and ``None`` valued
    ``file_id`` entries are dropped from citations.
    """
    formatted_block = {
        k: v
        for k, v in block.items()
        if k in ("type", "text", "cache_control", "citations")
    }
    citations = formatted_block.get("citations")
    if isinstance(citations, list):
        formatted_block["citations"] = [
            {
                k: v
                for k, v in citation.items()
                if not (k == "file_id" and v is None)
            }
            if isinstance(citation, dict)
            else citation
            for citation in citations
        ]
    return formatted_block


def _format_messages(
    messages: Sequence[BaseMessage],
    project: Optional[str] = None,
) -> tuple[Union[str, list[dict], None], list[dict]]:
    """Format messages for anthropic."""
    system: Union[str, list[dict], None] = None
    formatted_messages: list[dict] = []

    merged_messages = _merge_messages(messages)
    for i, message in enumerate(merged_messages):
        if message.type == "system":
            if i != 0:
                msg = "System message must be at beginning of message list."
                raise ValueError(msg)
            if isinstance(message.content, list):
                system = [
                    (
                        {"type": "text", "text": block}
                        if isinstance(block, str)
                        else (
                            _format_text_block(block)
                            if block.get("type") == "text"
                            else block
                        )
                    )
                    for block in message.content
                ]
            else:
                system = message.content
            continue

        role = _message_type_lookups[message.type]
        content: Union[str, list]

        if not isinstance(message.content, str):
            # parse as dict
            if not isinstance(message.content, list):
                msg = "Anthropic message content must be str or list of dicts"
                raise ValueError(msg)

            # populate content
            content = []
            for block in message.content:
                if isinstance(block, str):
                    content.append({"type": "text", "text": block})
                elif isinstance(block, dict):
                    if "type" not in block:
                        msg = "Dict content block must have a type key"
                        raise ValueError(msg)

                    if block["type"] == "text":
                        text = block.get("text", "")
                        # Only add non-empty strings for now as empty ones are
                        # invalid.
                        if text.strip():
                            content.append(_format_text_block(block))
                    else:
                        content.append(block)
                else:
                    msg = (
                        f"Content items must be str or dict, instead was: {type(block)}"
                    )
                    raise ValueError(msg)
        else:
            content = message.content

        formatted_messages.append({"role": role, "content": content})

    return system, formatted_messages

def test__format_messages_system_v1_content_blocks_drop_id() -> None:
    """v1 content blocks carry an ``id`` that Anthropic rejects on system blocks."""
    from langchain_core.messages.content_blocks import create_text_block

    system_block = dict(create_text_block("You are a helpful assistant.", id="lc_abc"))
    assert system_block["id"] == "lc_abc"

    messages = [
        SystemMessage(content=[system_block]),
        HumanMessage("Hello"),
    ]
    system, formatted_messages = _format_messages(messages)

    assert system == [{"type": "text", "text": "You are a helpful assistant."}]
    assert isinstance(system, list)
    assert "id" not in system[0]
    assert formatted_messages == [{"role": "user", "content": "Hello"}]


def test__format_messages_system_text_block_preserves_supported_fields() -> None:
    """Supported fields survive, unsupported ones are stripped."""
    messages = [
        SystemMessage(
            content=[
                {
                    "type": "text",
                    "text": "You are a helpful assistant.",
                    "cache_control": {"type": "ephemeral"},
                    "id": "lc_abc",
                }
            ]
        ),
        HumanMessage("Hello"),
    ]
    system, _ = _format_messages(messages)

    assert system == [
        {
            "type": "text",
            "text": "You are a helpful assistant.",
            "cache_control": {"type": "ephemeral"},
        }
    ]


def test__format_messages_system_non_text_blocks_pass_through() -> None:
    """Non-text system blocks are forwarded unchanged."""
    block = {
        "type": "search_result",
        "source": "https://example.com",
        "title": "Example",
        "content": [{"type": "text", "text": "hi"}],
    }
    messages = [SystemMessage(content=[block]), HumanMessage("Hello")]
    system, _ = _format_messages(messages)

    assert system == [block]
