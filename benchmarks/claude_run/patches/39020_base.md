def _get_tool_schema_for_token_counting(tool: Any) -> dict[str, Any]:
    """Get an OpenAI-style tool schema for approximate token counting.

    For ``BaseTool`` instances this reuses the cached ``tool_call_schema`` instead of
    recomputing the schema via ``convert_to_openai_tool``, which is significantly
    cheaper. The resulting schema can be slightly larger (it may retain ``$ref``,
    ``$defs`` and ``title`` fields), which is acceptable because token counting here
    is only an approximation.

    Args:
        tool: A dict, Pydantic model class, callable or ``BaseTool``.

    Returns:
        A dict representing the tool schema.
    """
    from langchain_core.tools import BaseTool  # avoid circular import

    if isinstance(tool, BaseTool):
        tool_call_schema = tool.tool_call_schema
        parameters = (
            tool_call_schema
            if isinstance(tool_call_schema, dict)
            else tool_call_schema.model_json_schema()
        )
        return {
            "type": "function",
            "function": {
                "name": tool.name,
                "description": tool.description,
                "parameters": parameters,
            },
        }
    return convert_to_openai_tool(tool)


def count_tokens_approximately(
    messages: Iterable[MessageLikeRepresentation],
    *,
    chars_per_token: float = 4.0,
    extra_tokens_per_message: float = 3.0,
    count_name: bool = True,
    tools: Sequence[dict[str, Any] | type | Callable | BaseTool] | None = None,
) -> int:
    """Approximate the total number of tokens in messages.

    The token count includes stringified message content, role, and (optionally) name.
    - For AI messages, the token count also includes stringified tool calls.
    - For tool messages, the token count also includes the tool call ID.
    - If ``tools`` is provided, the token count also includes the tool schemas.

    Args:
        messages: List of messages to count tokens for.
        chars_per_token: Number of characters per token to use for the approximation.
            Default is 4 (one token corresponds to ~4 chars for common English text).
            You can also specify float values for more fine-grained control.
            See more here: https://platform.openai.com/tokenizer
        extra_tokens_per_message: Number of extra tokens to add per message.
            Default is 3 (special tokens, including beginning/end of message).
            You can also specify float values for more fine-grained control.
            See more here:
            https://cookbook.openai.com/examples/how_to_count_tokens_with_tiktoken
        count_name: Whether to include message names in the count.
            Enabled by default.
        tools: Optional sequence of tools (dicts, Pydantic model classes, callables or
            ``BaseTool`` instances) whose schemas should be included in the count.

    Returns:
        Approximate number of tokens in the messages and tools.

    Note:
        This is a simple approximation that may differ from the actual token count
        returned by specific model providers.
    """
    token_count = 0.0
    for message in convert_to_messages(messages):
        message_chars = 0
        if isinstance(message.content, str):
            message_chars += len(message.content)

        # TODO: add support for approximate counting for image blocks
        else:
            content = repr(message.content)
            message_chars += len(content)

        if (
            isinstance(message, AIMessage)
            # exclude Anthropic format as tool calls are already included in the content
            and not isinstance(message.content, list)
            and message.tool_calls
        ):
            tool_calls_content = repr(message.tool_calls)
            message_chars += len(tool_calls_content)

        if isinstance(message, ToolMessage):
            message_chars += len(message.tool_call_id)

        role = _get_message_openai_role(message)
        message_chars += len(role)

        if message.name and count_name:
            message_chars += len(message.name)

        # NOTE: we're rounding up per message to ensure that
        # individual message token counts add up to the total count
        # for a list of messages
        token_count += math.ceil(message_chars / chars_per_token)

        # add extra tokens per message
        token_count += extra_tokens_per_message

    if tools:
        for tool in tools:
            tool_chars = len(json.dumps(_get_tool_schema_for_token_counting(tool)))
            token_count += math.ceil(tool_chars / chars_per_token)

    # round up once more time in case extra_tokens_per_message is a float
    return math.ceil(token_count)

def test_count_tokens_approximately_base_tool_uses_tool_call_schema() -> None:
    """``BaseTool`` token counting should reuse the cached ``tool_call_schema``."""
    from langchain_core.tools import tool

    @tool
    def my_tool(a: int, b: str) -> str:
        """Repeat b a times."""
        return b * a

    tool_call_schema = my_tool.tool_call_schema
    assert not isinstance(tool_call_schema, dict)
    expected_chars = len(
        json.dumps(
            {
                "type": "function",
                "function": {
                    "name": my_tool.name,
                    "description": my_tool.description,
                    "parameters": tool_call_schema.model_json_schema(),
                },
            }
        )
    )

    baseline = count_tokens_approximately([])
    with_tools = count_tokens_approximately([], tools=[my_tool])
    assert with_tools - baseline == math.ceil(expected_chars / 4.0)


def test_count_tokens_approximately_base_tool_does_not_call_converter(
    monkeypatch: pytest.MonkeyPatch,
) -> None:
    """``convert_to_openai_tool`` must not be called for ``BaseTool`` instances."""
    from langchain_core.tools import tool

    @tool
    def my_tool(a: int) -> int:
        """Double a."""
        return a * 2

    def _fail(*args: Any, **kwargs: Any) -> NoReturn:
        msg = "convert_to_openai_tool should not be called for BaseTool"
        raise AssertionError(msg)

    monkeypatch.setattr(
        "langchain_core.messages.utils.convert_to_openai_tool", _fail
    )

    assert count_tokens_approximately([], tools=[my_tool]) > 0


def test_count_tokens_approximately_base_tool_with_dict_schema() -> None:
    """``tool_call_schema`` that is already a raw dict should be handled."""
    from langchain_core.tools import StructuredTool

    json_schema = {
        "title": "dict_tool",
        "type": "object",
        "properties": {"x": {"type": "integer"}},
        "required": ["x"],
    }
    dict_tool = StructuredTool.from_function(
        func=lambda x: x,
        name="dict_tool",
        description="A tool with a dict args schema.",
        args_schema=json_schema,
    )

    tool_call_schema = dict_tool.tool_call_schema
    assert isinstance(tool_call_schema, dict)
    expected_chars = len(
        json.dumps(
            {
                "type": "function",
                "function": {
                    "name": dict_tool.name,
                    "description": dict_tool.description,
                    "parameters": tool_call_schema,
                },
            }
        )
    )

    baseline = count_tokens_approximately([])
    with_tools = count_tokens_approximately([], tools=[dict_tool])
    assert with_tools - baseline == math.ceil(expected_chars / 4.0)


def test_count_tokens_approximately_non_base_tool_unchanged() -> None:
    """Dict tools continue to go through ``convert_to_openai_tool``."""
    dict_tool = {
        "type": "function",
        "function": {
            "name": "get_weather",
            "description": "Get the weather.",
            "parameters": {
                "type": "object",
                "properties": {"city": {"type": "string"}},
                "required": ["city"],
            },
        },
    }

    expected_chars = len(json.dumps(convert_to_openai_tool(dict_tool)))
    baseline = count_tokens_approximately([])
    with_tools = count_tokens_approximately([], tools=[dict_tool])
    assert with_tools - baseline == math.ceil(expected_chars / 4.0)
