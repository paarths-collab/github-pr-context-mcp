```python
def test_gpt_5_3_chat_latest_profile_omits_reasoning_effort_levels() -> None:
    """gpt-5.3-chat-latest is non-reasoning: it must not advertise effort levels."""
    llm = ChatOpenAI(model="gpt-5.3-chat-latest", api_key="test")  # type: ignore[call-arg]
    profile = llm.profile
    assert profile is not None
    assert profile.get("reasoning_output") is False
    assert not profile.get("reasoning_effort_levels")
```
