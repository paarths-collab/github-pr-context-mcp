# libs/partners/openai/tests/unit_tests/chat_models/test_base.py

```python
def test_gpt_5_3_chat_latest_profile_omits_reasoning_effort() -> None:
    """`gpt-5.3-chat-latest` is GPT-5.3 Instant and takes no reasoning effort."""
    profile = ChatOpenAI(model="gpt-5.3-chat-latest", api_key="foo").profile  # type: ignore[arg-type]
    assert profile is not None
    assert profile.get("reasoning_output") is not True
    assert "reasoning_effort_levels" not in profile
    assert "reasoning_effort_default" not in profile


def test_reasoning_effort_levels_imply_reasoning_output() -> None:
    """Only reasoning models may advertise configurable reasoning effort."""
    from langchain_openai.data._profiles import PROFILES

    for model_name, profile in PROFILES.items():
        if "reasoning_effort_levels" in profile or "reasoning_effort_default" in profile:
            assert profile.get("reasoning_output") is True, model_name
```
