def _update_token_usage(
    overall_token_usage: dict[str, Any], new_usage: dict[str, Any]
) -> dict[str, Any]:
    """Recursively merge a token usage dict into an accumulator.

    Fireworks returns nested token usage details (e.g. ``prompt_tokens_details``)
    when prompt caching is involved. Summing those nested dicts with ``+`` raises
    ``TypeError``, so merge them recursively and only add numeric leaves.
    """
    for key, value in new_usage.items():
        if value is None:
            continue
        existing = overall_token_usage.get(key)
        if isinstance(value, dict):
            overall_token_usage[key] = _update_token_usage(
                existing if isinstance(existing, dict) else {}, value
            )
        elif isinstance(existing, (int, float)) and isinstance(value, (int, float)):
            overall_token_usage[key] = existing + value
        else:
            overall_token_usage[key] = value
    return overall_token_usage


def _create_usage_metadata(token_usage: dict[str, Any]) -> UsageMetadata:
    """Build standard usage metadata, normalizing Fireworks cached prompt tokens."""
    input_tokens = token_usage.get("prompt_tokens") or 0
    output_tokens = token_usage.get("completion_tokens") or 0
    total_tokens = token_usage.get("total_tokens") or input_tokens + output_tokens

    prompt_tokens_details = token_usage.get("prompt_tokens_details") or {}
    cache_read = prompt_tokens_details.get("cached_tokens") or 0

    usage_metadata: UsageMetadata = {
        "input_tokens": input_tokens,
        "output_tokens": output_tokens,
        "total_tokens": total_tokens,
    }
    if cache_read:
        input_token_details: InputTokenDetails = {"cache_read": cache_read}
        usage_metadata["input_token_details"] = input_token_details
    return usage_metadata


# Methods on ChatFireworks:

    def _combine_llm_outputs(self, llm_outputs: list[Optional[dict]]) -> dict:
        overall_token_usage: dict = {}
        system_fingerprint = None
        for output in llm_outputs:
            if output is None:
                # Happens in streaming
                continue
            token_usage = output["token_usage"]
            if token_usage is not None:
                _update_token_usage(overall_token_usage, token_usage)
            if system_fingerprint is None:
                system_fingerprint = output.get("system_fingerprint")
        combined = {"token_usage": overall_token_usage, "model_name": self.model_name}
        if system_fingerprint:
            combined["system_fingerprint"] = system_fingerprint
        return combined

    def _create_chat_result(self, response: Union[dict, BaseModel]) -> ChatResult:
        generations = []
        if not isinstance(response, dict):
            response = response.dict()
        token_usage = response.get("usage", {})
        for res in response["choices"]:
            message = _convert_dict_to_message(res["message"])
            if token_usage and isinstance(message, AIMessage):
                message.usage_metadata = _create_usage_metadata(token_usage)
            generation_info = {"finish_reason": res.get("finish_reason")}
            if "logprobs" in res:
                generation_info["logprobs"] = res["logprobs"]
            gen = ChatGeneration(
                message=message,
                generation_info=generation_info,
            )
            generations.append(gen)
        llm_output = {
            "token_usage": token_usage,
            "model_name": self.model_name,
            "system_fingerprint": response.get("system_fingerprint", ""),
        }
        return ChatResult(generations=generations, llm_output=llm_output)

def test_combine_llm_outputs_merges_nested_token_usage() -> None:
    """Nested token usage dicts should merge instead of raising TypeError."""
    llm = ChatFireworks(api_key="test", model="accounts/fireworks/models/foo")  # type: ignore[call-arg]
    llm_outputs: list = [
        {
            "token_usage": {
                "prompt_tokens": 10,
                "completion_tokens": 2,
                "total_tokens": 12,
                "prompt_tokens_details": {"cached_tokens": 8},
            },
            "model_name": llm.model_name,
        },
        {
            "token_usage": {
                "prompt_tokens": 20,
                "completion_tokens": 3,
                "total_tokens": 23,
                "prompt_tokens_details": {"cached_tokens": 16},
            },
            "model_name": llm.model_name,
        },
    ]

    combined = llm._combine_llm_outputs(llm_outputs)

    assert combined["token_usage"] == {
        "prompt_tokens": 30,
        "completion_tokens": 5,
        "total_tokens": 35,
        "prompt_tokens_details": {"cached_tokens": 24},
    }


def test_create_chat_result_reports_cached_prompt_tokens() -> None:
    """Cached prompt tokens are surfaced as ``input_token_details.cache_read``."""
    llm = ChatFireworks(api_key="test", model="accounts/fireworks/models/foo")  # type: ignore[call-arg]
    response = {
        "id": "chatcmpl-1",
        "object": "chat.completion",
        "created": 1700000000,
        "model": "accounts/fireworks/models/foo",
        "choices": [
            {
                "index": 0,
                "message": {"role": "assistant", "content": "hi"},
                "finish_reason": "stop",
            }
        ],
        "usage": {
            "prompt_tokens": 100,
            "completion_tokens": 5,
            "total_tokens": 105,
            "prompt_tokens_details": {"cached_tokens": 80},
        },
    }

    result = llm._create_chat_result(response)
    message = result.generations[0].message

    assert isinstance(message, AIMessage)
    assert message.usage_metadata is not None
    assert message.usage_metadata["input_tokens"] == 100
    assert message.usage_metadata["output_tokens"] == 5
    assert message.usage_metadata["total_tokens"] == 105
    assert message.usage_metadata["input_token_details"] == {"cache_read": 80}
