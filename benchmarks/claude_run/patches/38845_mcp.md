    def _should_retry(self, exc: Exception) -> bool:
        """Return whether `exc` is covered by `retry_on`.

        Exceptions that are not covered are neither retried nor normalized to a
        `ToolMessage`; they propagate unchanged so that other middleware (e.g.
        `ToolErrorMiddleware`) or the caller can handle them.
        """
        if isinstance(self.retry_on, tuple):
            return isinstance(exc, self.retry_on)
        return self.retry_on(exc)

    def wrap_tool_call(
        self,
        request: ToolCallRequest,
        handler: Callable[[ToolCallRequest], ToolMessage | Command],
    ) -> ToolMessage | Command:
        """Retry the tool call on retryable failures."""
        attempt = 0
        while True:
            try:
                return handler(request)
            except GraphBubbleUp:
                # Control-flow signals (interrupts, etc.) are never retried.
                raise
            except Exception as exc:
                # Only exceptions listed in `retry_on` are handled by this
                # middleware. Anything else propagates unchanged, including
                # when `on_failure="continue"`.
                if not self._should_retry(exc):
                    raise
                if attempt >= self.max_retries:
                    return self._handle_failure(exc, request)
                time.sleep(self._calculate_delay(attempt))
                attempt += 1

    async def awrap_tool_call(
        self,
        request: ToolCallRequest,
        handler: Callable[[ToolCallRequest], Awaitable[ToolMessage | Command]],
    ) -> ToolMessage | Command:
        """Retry the tool call on retryable failures."""
        attempt = 0
        while True:
            try:
                return await handler(request)
            except GraphBubbleUp:
                # Control-flow signals (interrupts, etc.) are never retried.
                raise
            except Exception as exc:
                # Only exceptions listed in `retry_on` are handled by this
                # middleware. Anything else propagates unchanged, including
                # when `on_failure="continue"`.
                if not self._should_retry(exc):
                    raise
                if attempt >= self.max_retries:
                    return self._handle_failure(exc, request)
                await asyncio.sleep(self._calculate_delay(attempt))
                attempt += 1

class _RetryableError(Exception):
    """Exception listed in `retry_on`."""


class _NonRetryableError(Exception):
    """Exception deliberately not listed in `retry_on`."""


def test_non_retryable_exception_propagates_with_on_failure_continue() -> None:
    """`on_failure="continue"` must not swallow exceptions outside `retry_on`."""
    calls = 0

    @tool
    def failing_tool(query: str) -> str:
        """Raise an exception that is not registered in `retry_on`."""
        nonlocal calls
        calls += 1
        msg = "not retryable"
        raise _NonRetryableError(msg)

    middleware = ToolRetryMiddleware(
        max_retries=2,
        initial_delay=0.0,
        jitter=False,
        retry_on=(_RetryableError,),
        on_failure="continue",
    )
    agent = create_agent(
        model=FakeToolCallingModel(
            tool_calls=[[{"name": "failing_tool", "args": {"query": "x"}, "id": "1"}], []]
        ),
        tools=[failing_tool],
        middleware=[middleware],
    )

    with pytest.raises(_NonRetryableError):
        agent.invoke({"messages": [HumanMessage("hi")]})

    # Non-retryable exceptions are raised on the first attempt, not retried.
    assert calls == 1


def test_retryable_exception_still_becomes_tool_message_with_on_failure_continue() -> None:
    """Exceptions in `retry_on` are still normalized to a `ToolMessage`."""
    calls = 0

    @tool
    def failing_tool(query: str) -> str:
        """Raise an exception that is registered in `retry_on`."""
        nonlocal calls
        calls += 1
        msg = "retryable"
        raise _RetryableError(msg)

    middleware = ToolRetryMiddleware(
        max_retries=2,
        initial_delay=0.0,
        jitter=False,
        retry_on=(_RetryableError,),
        on_failure="continue",
    )
    agent = create_agent(
        model=FakeToolCallingModel(
            tool_calls=[[{"name": "failing_tool", "args": {"query": "x"}, "id": "1"}], []]
        ),
        tools=[failing_tool],
        middleware=[middleware],
    )

    result = agent.invoke({"messages": [HumanMessage("hi")]})

    tool_messages = [m for m in result["messages"] if isinstance(m, ToolMessage)]
    assert len(tool_messages) == 1
    assert tool_messages[0].status == "error"
    assert calls == 3
