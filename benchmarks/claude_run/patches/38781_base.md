"""Middleware that converts tool execution errors into `ToolMessage`s."""

from __future__ import annotations

from collections.abc import Awaitable, Callable
from typing import TYPE_CHECKING

from langchain_core.messages import ToolMessage
from langgraph.errors import GraphBubbleUp

from langchain.agents.middleware.types import AgentMiddleware, ToolCallRequest

if TYPE_CHECKING:
    from langgraph.types import Command

__all__ = ["ToolErrorMiddleware"]

DEFAULT_TOOL_ERROR_MESSAGE = "Error: {error!r}\nPlease fix your mistakes."

ToolErrorHandler = Callable[[Exception, ToolCallRequest], "str | ToolMessage | None"]


def _default_on_error(exc: Exception, request: ToolCallRequest) -> str:  # noqa: ARG001
    return DEFAULT_TOOL_ERROR_MESSAGE.format(error=exc)


class ToolErrorMiddleware(AgentMiddleware):
    """Catch tool execution errors and return them to the model as `ToolMessage`s.

    The handler receives the raised exception and the originating
    [`ToolCallRequest`][langchain.agents.middleware.types.ToolCallRequest]. Return a
    string (or a fully formed `ToolMessage`) to send the error back to the model, or
    return `None` to let the exception propagate.

    Example:
        ```python
        def on_error(exc: Exception, request: ToolCallRequest) -> str | None:
            if isinstance(exc, MyError):
                return f"Tool '{request.tool_call['name']}' failed with {type(exc).__name__}"
            # propagates everything else
            return None


        agent = create_agent(
            model="...",
            tools=[failing_tool],
            middleware=[ToolErrorMiddleware(on_error)],
        )
        ```
    """

    def __init__(
        self,
        on_error: str | ToolErrorHandler = _default_on_error,
        *,
        handled_exceptions: tuple[type[Exception], ...] = (Exception,),
    ) -> None:
        """Initialize the middleware.

        Args:
            on_error: Either a static string to return as the tool result, or a callable
                accepting `(exception, request)` that returns a string, a `ToolMessage`,
                or `None` to re-raise the exception.
            handled_exceptions: Exception types eligible for handling. Anything else
                propagates untouched.
        """
        super().__init__()
        self.on_error = on_error
        self.handled_exceptions = handled_exceptions

    def _handle(self, exc: Exception, request: ToolCallRequest) -> ToolMessage:
        """Convert `exc` into a `ToolMessage`, or re-raise it if unhandled."""
        if not isinstance(exc, self.handled_exceptions):
            raise exc

        result = self.on_error(exc, request) if callable(self.on_error) else self.on_error
        if result is None:
            raise exc
        if isinstance(result, ToolMessage):
            return result
        return ToolMessage(
            content=result,
            name=request.tool_call["name"],
            tool_call_id=request.tool_call["id"],
            status="error",
        )

    def wrap_tool_call(
        self,
        request: ToolCallRequest,
        handler: Callable[[ToolCallRequest], ToolMessage | Command],
    ) -> ToolMessage | Command:
        """Execute the tool call, translating errors into `ToolMessage`s."""
        try:
            return handler(request)
        except GraphBubbleUp:
            # Control-flow signals (interrupts, etc.) must never be swallowed.
            raise
        except Exception as exc:  # noqa: BLE001
            return self._handle(exc, request)

    async def awrap_tool_call(
        self,
        request: ToolCallRequest,
        handler: Callable[[ToolCallRequest], Awaitable[ToolMessage | Command]],
    ) -> ToolMessage | Command:
        """Async variant of `wrap_tool_call`."""
        try:
            return await handler(request)
        except GraphBubbleUp:
            raise
        except Exception as exc:  # noqa: BLE001
            return self._handle(exc, request)
```

"""Unit tests for `ToolErrorMiddleware`."""

from types import SimpleNamespace
from typing import cast

import pytest
from langchain_core.messages import AIMessage, ToolMessage
from langchain_core.tools import tool
from langgraph.errors import GraphInterrupt

from langchain.agents import create_agent
from langchain.agents.middleware import ToolErrorMiddleware
from langchain.agents.middleware.types import ToolCallRequest

from ...model import FakeToolCallingModel


class MyError(Exception):
    """Error the handler knows how to translate."""


class OtherError(Exception):
    """Error the handler does not know how to translate."""


def _request(name: str = "failing_tool", id_: str = "call_1") -> ToolCallRequest:
    return cast(
        ToolCallRequest,
        SimpleNamespace(tool_call={"name": name, "args": {}, "id": id_}),
    )


def _raising_handler(exc: Exception):
    def handler(request: ToolCallRequest) -> ToolMessage:
        raise exc

    return handler


def test_handled_error_becomes_tool_message() -> None:
    def on_error(exc: Exception, request: ToolCallRequest) -> str | None:
        if isinstance(exc, MyError):
            return f"Tool '{request.tool_call['name']}' failed with {type(exc).__name__}"
        return None

    middleware = ToolErrorMiddleware(on_error)
    result = middleware.wrap_tool_call(_request(), _raising_handler(MyError("boom")))

    assert isinstance(result, ToolMessage)
    assert result.content == "Tool 'failing_tool' failed with MyError"
    assert result.tool_call_id == "call_1"
    assert result.name == "failing_tool"
    assert result.status == "error"


def test_unhandled_error_propagates() -> None:
    def on_error(exc: Exception, request: ToolCallRequest) -> str | None:
        if isinstance(exc, MyError):
            return "handled"
        return None

    middleware = ToolErrorMiddleware(on_error)
    with pytest.raises(OtherError):
        middleware.wrap_tool_call(_request(), _raising_handler(OtherError("nope")))


def test_handled_exceptions_filter() -> None:
    middleware = ToolErrorMiddleware("handled", handled_exceptions=(MyError,))

    result = middleware.wrap_tool_call(_request(), _raising_handler(MyError("boom")))
    assert isinstance(result, ToolMessage)
    assert result.content == "handled"

    with pytest.raises(OtherError):
        middleware.wrap_tool_call(_request(), _raising_handler(OtherError("nope")))


def test_string_on_error() -> None:
    middleware = ToolErrorMiddleware("static message")
    result = middleware.wrap_tool_call(_request(), _raising_handler(MyError("boom")))
    assert isinstance(result, ToolMessage)
    assert result.content == "static message"


def test_default_on_error() -> None:
    middleware = ToolErrorMiddleware()
    result = middleware.wrap_tool_call(_request(), _raising_handler(MyError("boom")))
    assert isinstance(result, ToolMessage)
    assert "boom" in result.content


def test_tool_message_passthrough() -> None:
    message = ToolMessage(content="custom", tool_call_id="call_1", name="failing_tool")
    middleware = ToolErrorMiddleware(lambda exc, request: message)
    result = middleware.wrap_tool_call(_request(), _raising_handler(MyError("boom")))
    assert result is message


def test_success_is_not_intercepted() -> None:
    message = ToolMessage(content="ok", tool_call_id="call_1", name="failing_tool")
    middleware = ToolErrorMiddleware("handled")
    result = middleware.wrap_tool_call(_request(), lambda request: message)
    assert result is message


def test_graph_bubble_up_is_not_swallowed() -> None:
    middleware = ToolErrorMiddleware("handled")
    with pytest.raises(GraphInterrupt):
        middleware.wrap_tool_call(_request(), _raising_handler(GraphInterrupt(())))


async def test_async_handled_error_becomes_tool_message() -> None:
    async def handler(request: ToolCallRequest) -> ToolMessage:
        raise MyError("boom")

    middleware = ToolErrorMiddleware(
        lambda exc, request: f"Tool '{request.tool_call['name']}' failed with {type(exc).__name__}"
    )
    result = await middleware.awrap_tool_call(_request(), handler)

    assert isinstance(result, ToolMessage)
    assert result.content == "Tool 'failing_tool' failed with MyError"


async def test_async_unhandled_error_propagates() -> None:
    async def handler(request: ToolCallRequest) -> ToolMessage:
        raise OtherError("nope")

    middleware = ToolErrorMiddleware(lambda exc, request: None)
    with pytest.raises(OtherError):
        await middleware.awrap_tool_call(_request(), handler)


def test_agent_integration() -> None:
    @tool
    def failing_tool(x: int) -> str:
        """Always fails."""
        raise MyError("boom")

    def on_error(exc: Exception, request: ToolCallRequest) -> str | None:
        if isinstance(exc, MyError):
            return f"Tool '{request.tool_call['name']}' failed with {type(exc).__name__}"
        return None

    model = FakeToolCallingModel(
        tool_calls=[[{"name": "failing_tool", "args": {"x": 1}, "id": "call_1"}], []]
    )
    agent = create_agent(
        model=model,
        tools=[failing_tool],
        middleware=[ToolErrorMiddleware(on_error)],
    )

    result = agent.invoke({"messages": [{"role": "user", "content": "go"}]})
    tool_messages = [m for m in result["messages"] if isinstance(m, ToolMessage)]

    assert len(tool_messages) == 1
    assert tool_messages[0].content == "Tool 'failing_tool' failed with MyError"
    assert isinstance(result["messages"][-1], AIMessage)
