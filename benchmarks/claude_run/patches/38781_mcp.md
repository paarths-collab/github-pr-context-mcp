"""Middleware that converts tool execution errors into `ToolMessage`s."""

from __future__ import annotations

import inspect
from collections.abc import Awaitable, Callable
from typing import TYPE_CHECKING, Union

from langchain_core.messages import ToolMessage
from langgraph.errors import GraphBubbleUp

from langchain.agents.middleware.types import AgentMiddleware, ToolCallRequest

if TYPE_CHECKING:
    from langgraph.types import Command

__all__ = ["ToolErrorMiddleware"]

ToolErrorHandlerResult = Union[str, ToolMessage, None]
"""Result of a tool error handler.

- `str`: content of the `ToolMessage` returned in place of the error.
- `ToolMessage`: returned as-is.
- `None`: the exception is propagated.
"""

ToolErrorHandler = Callable[
    [Exception, ToolCallRequest],
    Union[ToolErrorHandlerResult, Awaitable[ToolErrorHandlerResult]],
]


def _default_on_error(exc: Exception, request: ToolCallRequest) -> str:
    """Return a generic error string for any exception."""
    return f"Error: {exc!r}\n Please fix your mistakes."


class ToolErrorMiddleware(AgentMiddleware):
    """Catch tool execution errors and return them to the model as `ToolMessage`s.

    Args:
        on_error: How to handle an exception raised by a tool. Either a static
            string used as the `ToolMessage` content, or a callable receiving the
            exception and the `ToolCallRequest`. The callable may return a string
            (used as `ToolMessage` content), a `ToolMessage` (returned as-is), or
            `None` to re-raise the exception. Async callables are supported when
            the agent is invoked asynchronously.

    Example:
        ```python
        def on_error(exc: Exception, request: ToolCallRequest) -> str | None:
            if isinstance(exc, MyError):
                return f"Tool '{request.tool_call['name']}' failed with {type(exc).__name__}"
            # propagates everything else


        agent = create_agent(
            model="...",
            tools=[failing_tool],
            middleware=[ToolErrorMiddleware(on_error)],
        )
        ```
    """

    def __init__(self, on_error: str | ToolErrorHandler = _default_on_error) -> None:
        """Initialize the middleware with an error handler."""
        super().__init__()
        self.on_error = on_error

    def _to_message(
        self, result: ToolErrorHandlerResult, request: ToolCallRequest
    ) -> ToolMessage | None:
        """Coerce a handler result into a `ToolMessage`, or `None` to propagate."""
        if result is None:
            return None
        if isinstance(result, ToolMessage):
            return result
        return ToolMessage(
            content=result,
            name=request.tool_call["name"],
            tool_call_id=request.tool_call["id"],
            status="error",
        )

    def wrap_tool_call(
        self,
        request: ToolCallRequest,
        handler: Callable[[ToolCallRequest], ToolMessage | Command],
    ) -> ToolMessage | Command:
        """Execute the tool call, converting handled exceptions into `ToolMessage`s."""
        try:
            return handler(request)
        except GraphBubbleUp:
            # Control flow (e.g. interrupts) is not a tool error.
            raise
        except Exception as exc:  # noqa: BLE001
            if isinstance(self.on_error, str):
                result: ToolErrorHandlerResult = self.on_error
            else:
                result = self.on_error(exc, request)  # type: ignore[assignment]
                if inspect.isawaitable(result):
                    msg = (
                        "Async `on_error` handlers are only supported when the agent "
                        "is invoked asynchronously."
                    )
                    raise TypeError(msg) from exc
            message = self._to_message(result, request)
            if message is None:
                raise
            return message

    async def awrap_tool_call(
        self,
        request: ToolCallRequest,
        handler: Callable[[ToolCallRequest], Awaitable[ToolMessage | Command]],
    ) -> ToolMessage | Command:
        """Async version of `wrap_tool_call`."""
        try:
            return await handler(request)
        except GraphBubbleUp:
            raise
        except Exception as exc:  # noqa: BLE001
            if isinstance(self.on_error, str):
                result: ToolErrorHandlerResult = self.on_error
            else:
                maybe_awaitable = self.on_error(exc, request)
                result = (
                    await maybe_awaitable
                    if inspect.isawaitable(maybe_awaitable)
                    else maybe_awaitable  # type: ignore[assignment]
                )
            message = self._to_message(result, request)
            if message is None:
                raise
            return message

```python
"""Unit tests for `ToolErrorMiddleware`."""

import pytest
from langchain_core.messages import HumanMessage, ToolMessage
from langchain_core.tools import tool

from langchain.agents import create_agent
from langchain.agents.middleware import ToolErrorMiddleware
from langchain.agents.middleware.types import ToolCallRequest

from ...model import FakeToolCallingModel


class MyError(Exception):
    """Error raised by the failing test tool."""


class OtherError(Exception):
    """Error the handler does not recognize."""


@tool
def failing_tool(x: int) -> str:
    """Always fails with MyError."""
    msg = "boom"
    raise MyError(msg)


@tool
def other_failing_tool(x: int) -> str:
    """Always fails with OtherError."""
    msg = "other boom"
    raise OtherError(msg)


def _model(tool_name: str) -> FakeToolCallingModel:
    return FakeToolCallingModel(
        tool_calls=[[{"name": tool_name, "args": {"x": 1}, "id": "call_1"}], []]
    )


def _tool_messages(result: dict) -> list[ToolMessage]:
    return [m for m in result["messages"] if isinstance(m, ToolMessage)]


def test_callable_handler_converts_error_to_tool_message() -> None:
    def on_error(exc: Exception, request: ToolCallRequest) -> str | None:
        if isinstance(exc, MyError):
            return f"Tool '{request.tool_call['name']}' failed with {type(exc).__name__}"
        return None

    agent = create_agent(
        model=_model("failing_tool"),
        tools=[failing_tool],
        middleware=[ToolErrorMiddleware(on_error)],
    )
    result = agent.invoke({"messages": [HumanMessage("hi")]})

    messages = _tool_messages(result)
    assert len(messages) == 1
    assert messages[0].content == "Tool 'failing_tool' failed with MyError"
    assert messages[0].tool_call_id == "call_1"
    assert messages[0].name == "failing_tool"
    assert messages[0].status == "error"


def test_handler_returning_none_propagates() -> None:
    def on_error(exc: Exception, request: ToolCallRequest) -> str | None:
        if isinstance(exc, MyError):
            return "handled"
        return None

    agent = create_agent(
        model=_model("other_failing_tool"),
        tools=[other_failing_tool],
        middleware=[ToolErrorMiddleware(on_error)],
    )
    with pytest.raises(OtherError):
        agent.invoke({"messages": [HumanMessage("hi")]})


def test_static_string_handler() -> None:
    agent = create_agent(
        model=_model("failing_tool"),
        tools=[failing_tool],
        middleware=[ToolErrorMiddleware("tool unavailable")],
    )
    result = agent.invoke({"messages": [HumanMessage("hi")]})

    messages = _tool_messages(result)
    assert messages[0].content == "tool unavailable"
    assert messages[0].status == "error"


def test_default_handler_catches_all_errors() -> None:
    agent = create_agent(
        model=_model("failing_tool"),
        tools=[failing_tool],
        middleware=[ToolErrorMiddleware()],
    )
    result = agent.invoke({"messages": [HumanMessage("hi")]})

    messages = _tool_messages(result)
    assert len(messages) == 1
    assert "MyError" in messages[0].content
    assert messages[0].status == "error"


def test_handler_returning_tool_message_is_used_as_is() -> None:
    def on_error(exc: Exception, request: ToolCallRequest) -> ToolMessage:
        return ToolMessage(
            content="custom",
            name=request.tool_call["name"],
            tool_call_id=request.tool_call["id"],
        )

    agent = create_agent(
        model=_model("failing_tool"),
        tools=[failing_tool],
        middleware=[ToolErrorMiddleware(on_error)],
    )
    result = agent.invoke({"messages": [HumanMessage("hi")]})

    messages = _tool_messages(result)
    assert messages[0].content == "custom"
    assert messages[0].status == "success"


def test_successful_tool_call_is_untouched() -> None:
    @tool
    def working_tool(x: int) -> str:
        """Returns a value."""
        return "ok"

    agent = create_agent(
        model=FakeToolCallingModel(
            tool_calls=[[{"name": "working_tool", "args": {"x": 1}, "id": "call_1"}], []]
        ),
        tools=[working_tool],
        middleware=[ToolErrorMiddleware()],
    )
    result = agent.invoke({"messages": [HumanMessage("hi")]})

    messages = _tool_messages(result)
    assert messages[0].content == "ok"
    assert messages[0].status == "success"


async def test_async_handler() -> None:
    async def on_error(exc: Exception, request: ToolCallRequest) -> str | None:
        if isinstance(exc, MyError):
            return "async handled"
        return None

    agent = create_agent(
        model=_model("failing_tool"),
        tools=[failing_tool],
        middleware=[ToolErrorMiddleware(on_error)],
    )
    result = await agent.ainvoke({"messages": [HumanMessage("hi")]})

    messages = _tool_messages(result)
    assert messages[0].content == "async handled"
    assert messages[0].status == "error"


async def test_async_handler_returning_none_propagates() -> None:
    async def on_error(exc: Exception, request: ToolCallRequest) -> str | None:
        return None

    agent = create_agent(
        model=_model("failing_tool"),
        tools=[failing_tool],
        middleware=[ToolErrorMiddleware(on_error)],
    )
    with pytest.raises(MyError):
        await agent.ainvoke({"messages": [HumanMessage("hi")]})


def test_tool_error_middleware_importable_from_public_module() -> None:
    from langchain.agents.middleware import ToolErrorMiddleware as PublicToolErrorMiddleware

    assert PublicToolErrorMiddleware is ToolErrorMiddleware
