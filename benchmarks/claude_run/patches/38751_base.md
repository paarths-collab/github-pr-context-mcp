```python
def _update_token_usage(
    overall_token_usage: Union[int, dict], new_usage: Union[int, dict]
) -> Union[int, dict]:
    """Recursively combine token usage values.

    Fireworks can return nested token usage dictionaries (for example
    ``prompt_tokens_details``), so scalar addition alone is not enough when
    combining the usage of several generations.
    """
    if isinstance(new_usage, int):
        if not isinstance(overall_token_usage, int):
            raise ValueError(
                f"Got different types for token usage: "
                f"{type(new_usage)} and {type(overall_token_usage)}"
            )
        return overall_token_usage + new_usage
    if isinstance(new_usage, dict):
        if not isinstance(overall_token_usage, dict):
            raise ValueError(
                f"Got different types for token usage: "
                f"{type(new_usage)} and {type(overall_token_usage)}"
            )
        return {
            k: _update_token_usage(overall_token_usage.get(k, 0), v)
            for k, v in new_usage.items()
        }
    warnings.warn(f"Unexpected type for token usage: {type(new_usage)}")
    return overall_token_usage


def _create_usage_metadata(token_usage: dict) -> UsageMetadata:
    """Build ``UsageMetadata`` from a Fireworks ``usage`` payload.

    Cached prompt tokens are normalized into
    ``input_token_details.cache_read`` so downstream tracing and cost
    reporting can distinguish cached input tokens from uncached ones.
    """
    input_tokens = token_usage.get("prompt_tokens") or 0
    output_tokens = token_usage.get("completion_tokens") or 0
    total_tokens = token_usage.get("total_tokens") or input_tokens + output_tokens

    prompt_tokens_details = token_usage.get("prompt_tokens_details") or {}
    if not isinstance(prompt_tokens_details, dict):
        prompt_tokens_details = {}
    cache_read = prompt_tokens_details.get("cached_tokens")
    if cache_read is None:
        cache_read = token_usage.get("cached_prompt_tokens")

    input_token_details: dict = {}
    if isinstance(cache_read, int):
        input_token_details["cache_read"] = cache_read

    usage_metadata = UsageMetadata(
        input_tokens=input_tokens,
        output_tokens=output_tokens,
        total_tokens=total_tokens,
    )
    if input_token_details:
        usage_metadata["input_token_details"] = InputTokenDetails(
            **input_token_details
        )
    return usage_metadata
```

```python
    def _combine_llm_outputs(self, llm_outputs: list[Optional[dict]]) -> dict:
        overall_token_usage: dict = {}
        system_fingerprint = None
        for output in llm_outputs:
            if output is None:
                # Happens in streaming
                continue
            token_usage = output["token_usage"]
            if token_usage is not None:
                for k, v in token_usage.items():
                    if k in overall_token_usage and overall_token_usage[k] is not None:
                        overall_token_usage[k] = _update_token_usage(
                            overall_token_usage[k], v
                        )
                    else:
                        overall_token_usage[k] = v
            if system_fingerprint is None:
                system_fingerprint = output.get("system_fingerprint")
        combined = {"token_usage": overall_token_usage, "model_name": self.model_name}
        if system_fingerprint:
            combined["system_fingerprint"] = system_fingerprint
        return combined
```

```python
    def _create_chat_result(self, response: Union[dict, BaseModel]) -> ChatResult:
        generations = []
        if not isinstance(response, dict):
            response = response.dict()
        token_usage = response.get("usage", {}) or {}
        for res in response["choices"]:
            message = _convert_dict_to_message(res["message"])
            if token_usage and isinstance(message, AIMessage):
                message.usage_metadata = _create_usage_metadata(token_usage)
            generation_info = dict(finish_reason=res.get("finish_reason"))
            if "logprobs" in res:
                generation_info["logprobs"] = res["logprobs"]
            gen = ChatGeneration(
                message=message,
                generation_info=generation_info,
            )
            generations.append(gen)
        llm_output = {
            "token_usage": token_usage,
            "model_name": self.model_name,
            "system_fingerprint": response.get("system_fingerprint", ""),
        }
        return ChatResult(generations=generations, llm_output=llm_output)
```

```python
def test_chat_fireworks_combine_llm_outputs_nested_token_usage() -> None:
    """Nested token usage dicts should merge instead of raising TypeError."""
    llm = ChatFireworks(api_key="test", model="accounts/fireworks/models/llama-v3-8b")
    llm_outputs = [
        {
            "token_usage": {
                "prompt_tokens": 10,
                "completion_tokens": 2,
                "total_tokens": 12,
                "prompt_tokens_details": {"cached_tokens": 4},
            },
            "model_name": llm.model_name,
        },
        {
            "token_usage": {
                "prompt_tokens": 20,
                "completion_tokens": 3,
                "total_tokens": 23,
                "prompt_tokens_details": {"cached_tokens": 8},
            },
            "model_name": llm.model_name,
        },
    ]

    combined = llm._combine_llm_outputs(llm_outputs)

    assert combined["token_usage"] == {
        "prompt_tokens": 30,
        "completion_tokens": 5,
        "total_tokens": 35,
        "prompt_tokens_details": {"cached_tokens": 12},
    }


def test_chat_fireworks_cached_prompt_tokens_usage_metadata() -> None:
    """Cached prompt tokens are surfaced as input_token_details.cache_read."""
    llm = ChatFireworks(api_key="test", model="accounts/fireworks/models/llama-v3-8b")
    response = {
        "id": "chatcmpl-1",
        "object": "chat.completion",
        "created": 1700000000,
        "model": "accounts/fireworks/models/llama-v3-8b",
        "choices": [
            {
                "index": 0,
                "message": {"role": "assistant", "content": "Hello"},
                "finish_reason": "stop",
            }
        ],
        "usage": {
            "prompt_tokens": 100,
            "completion_tokens": 5,
            "total_tokens": 105,
            "prompt_tokens_details": {"cached_tokens": 60},
        },
    }

    result = llm._create_chat_result(response)
    message = result.generations[0].message

    assert isinstance(message, AIMessage)
    assert message.usage_metadata is not None
    assert message.usage_metadata["input_tokens"] == 100
    assert message.usage_metadata["output_tokens"] == 5
    assert message.usage_metadata["total_tokens"] == 105
    assert message.usage_metadata["input_token_details"]["cache_read"] == 60
```
