def _format_text_block(block: dict[str, Any]) -> dict[str, Any]:
    """Narrow a text content block to the fields the Anthropic API accepts.

    LangChain v1 content blocks carry extra bookkeeping fields (e.g. ``id``) that
    Anthropic rejects with ``400 ... Extra inputs are not permitted``.
    """
    formatted_block = {
        k: v
        for k, v in block.items()
        if k in ("type", "text", "cache_control", "citations")
    }
    citations = formatted_block.get("citations")
    if isinstance(citations, dict) and citations.get("file_id") is None:
        formatted_block["citations"] = {
            k: v for k, v in citations.items() if k != "file_id"
        }
    return formatted_block

```python
def _format_messages(
    messages: Sequence[BaseMessage],
) -> tuple[Union[str, list[dict], None], list[dict]]:
    """Format messages for anthropic."""
    system: Union[str, list[dict], None] = None
    formatted_messages: list[dict] = []
    merged_messages = _merge_messages(messages)
    for i, message in enumerate(merged_messages):
        if message.type == "system":
            if system is not None:
                msg = "Received multiple non-consecutive system messages."
                raise ValueError(msg)
            if isinstance(message.content, list):
                system = [
                    (
                        _format_text_block(block)
                        if isinstance(block, dict) and block.get("type") == "text"
                        else (
                            block
                            if isinstance(block, dict)
                            else {"type": "text", "text": block}
                        )
                    )
                    for block in message.content
                ]
            else:
                system = message.content
            continue

        role = _message_type_lookups[message.type]
        content: Union[str, list]

        if not isinstance(message.content, str):
            # parse as dict
            if not isinstance(message.content, list):
                msg = "Anthropic message content must be str or list of dicts"
                raise ValueError(msg)

            # populate content
            content = []
            for block in message.content:
                if isinstance(block, str):
                    content.append({"type": "text", "text": block})
                elif isinstance(block, dict):
                    if "type" not in block:
                        msg = "Dict content block must have a type key"
                        raise ValueError(msg)

                    if block["type"] == "text":
                        text = block.get("text", "")
                        # Only add non-empty strings for now as empty ones are
                        # not accepted.
                        # https://github.com/anthropics/anthropic-sdk-python/issues/461
                        if text.strip():
                            content.append(_format_text_block(block))
                    else:
                        content.append(block)
                else:
                    msg = (
                        f"Content blocks must be str or dict, instead was: {type(block)}"
                    )
                    raise ValueError(msg)
        else:
            content = message.content

        formatted_messages.append({"role": role, "content": content})
    return system, formatted_messages
```

def test__format_messages_system_v1_content_blocks_drop_id() -> None:
    """v1 content blocks carry an ``id`` that Anthropic rejects on system blocks."""
    from langchain_core.messages.content_blocks import create_text_block

    block = create_text_block("You are a helpful assistant.")
    assert "id" in block

    messages = [SystemMessage(content=[block]), HumanMessage("hi")]
    expected_system = [{"type": "text", "text": "You are a helpful assistant."}]
    expected_messages = [{"role": "user", "content": "hi"}]

    actual_system, actual_messages = _format_messages(messages)
    assert actual_system == expected_system
    assert actual_messages == expected_messages


def test__format_messages_system_text_block_preserves_supported_fields() -> None:
    """Supported fields survive; unsupported ones are stripped."""
    messages = [
        SystemMessage(
            content=[
                {
                    "type": "text",
                    "text": "cached instructions",
                    "cache_control": {"type": "ephemeral"},
                    "id": "lc_abc123",
                },
                {"type": "not_text", "id": "lc_def456"},
            ]
        ),
        HumanMessage("hi"),
    ]
    expected_system = [
        {
            "type": "text",
            "text": "cached instructions",
            "cache_control": {"type": "ephemeral"},
        },
        {"type": "not_text", "id": "lc_def456"},
    ]
    expected_messages = [{"role": "user", "content": "hi"}]

    actual_system, actual_messages = _format_messages(messages)
    assert actual_system == expected_system
    assert actual_messages == expected_messages
