```python
    def _is_retryable(self, exc: Exception) -> bool:
        """Whether ``exc`` is covered by the configured ``retry_on``.

        Exceptions that are not covered are never handled by this middleware:
        they are neither retried nor normalized into a ``ToolMessage``.
        """
        if isinstance(self.retry_on, tuple):
            return isinstance(exc, self.retry_on)
        return bool(self.retry_on(exc))

    def _handle_exhausted(
        self,
        exc: Exception,
        request: ToolCallRequest,
    ) -> ToolMessage:
        """Produce the final result once all retries for a retryable ``exc`` are used up."""
        if self.on_failure == "error":
            raise exc

        if callable(self.on_failure):
            content = self.on_failure(exc)
        elif self.on_failure == "continue":
            content = (
                f"Error: {type(exc).__name__}: {exc}\n"
                f"Tool '{request.tool_call['name']}' failed after "
                f"{self.max_retries + 1} attempt(s)."
            )
        else:
            content = self.on_failure

        return ToolMessage(
            content=content,
            tool_call_id=request.tool_call["id"] or "",
            name=request.tool_call["name"],
            status="error",
        )

    def wrap_tool_call(
        self,
        request: ToolCallRequest,
        handler: Callable[[ToolCallRequest], ToolMessage | Command],
    ) -> ToolMessage | Command:
        """Retry the tool call on retryable failures.

        Only exceptions matched by ``retry_on`` are handled here. Any other
        exception propagates unchanged, so ``on_failure`` never converts an
        exception the user did not opt into retrying.
        """
        last_exception: Exception | None = None

        for attempt in range(self.max_retries + 1):
            try:
                return handler(request)
            except Exception as exc:
                if not self._is_retryable(exc):
                    raise
                last_exception = exc
                if attempt < self.max_retries:
                    time.sleep(self._calculate_delay(attempt))

        if last_exception is None:  # pragma: no cover - defensive
            msg = "Tool call failed without an exception."
            raise RuntimeError(msg)

        return self._handle_exhausted(last_exception, request)

    async def awrap_tool_call(
        self,
        request: ToolCallRequest,
        handler: Callable[[ToolCallRequest], Awaitable[ToolMessage | Command]],
    ) -> ToolMessage | Command:
        """Async version of `wrap_tool_call`.

        Only exceptions matched by ``retry_on`` are handled here. Any other
        exception propagates unchanged, so ``on_failure`` never converts an
        exception the user did not opt into retrying.
        """
        last_exception: Exception | None = None

        for attempt in range(self.max_retries + 1):
            try:
                return await handler(request)
            except Exception as exc:
                if not self._is_retryable(exc):
                    raise
                last_exception = exc
                if attempt < self.max_retries:
                    await asyncio.sleep(self._calculate_delay(attempt))

        if last_exception is None:  # pragma: no cover - defensive
            msg = "Tool call failed without an exception."
            raise RuntimeError(msg)

        return self._handle_exhausted(last_exception, request)
```

```python
class _RetryableError(Exception):
    """Error type registered in ``retry_on``."""


class _NonRetryableError(Exception):
    """Error type deliberately absent from ``retry_on``."""


def _make_request() -> ToolCallRequest:
    @tool
    def failing_tool() -> str:
        """A tool that always fails."""
        raise _NonRetryableError("boom")

    return ToolCallRequest(
        tool_call={
            "name": "failing_tool",
            "args": {},
            "id": "call_1",
            "type": "tool_call",
        },
        tool=failing_tool,
        state={"messages": []},
        runtime=None,  # type: ignore[arg-type]
    )


def test_non_retryable_exception_propagates_with_on_failure_continue() -> None:
    """Exceptions outside ``retry_on`` are not retried nor turned into ToolMessages."""
    middleware = ToolRetryMiddleware(
        max_retries=3,
        initial_delay=0.0,
        retry_on=(_RetryableError,),
        on_failure="continue",
    )
    attempts = 0

    def handler(request: ToolCallRequest) -> ToolMessage:
        nonlocal attempts
        attempts += 1
        raise _NonRetryableError("boom")

    with pytest.raises(_NonRetryableError, match="boom"):
        middleware.wrap_tool_call(_make_request(), handler)

    assert attempts == 1


async def test_non_retryable_exception_propagates_with_on_failure_continue_async() -> None:
    """Async version: exceptions outside ``retry_on`` propagate untouched."""
    middleware = ToolRetryMiddleware(
        max_retries=3,
        initial_delay=0.0,
        retry_on=(_RetryableError,),
        on_failure="continue",
    )
    attempts = 0

    async def handler(request: ToolCallRequest) -> ToolMessage:
        nonlocal attempts
        attempts += 1
        raise _NonRetryableError("boom")

    with pytest.raises(_NonRetryableError, match="boom"):
        await middleware.awrap_tool_call(_make_request(), handler)

    assert attempts == 1


def test_retryable_exception_becomes_tool_message_with_on_failure_continue() -> None:
    """Exceptions listed in ``retry_on`` are still normalized after retries."""
    middleware = ToolRetryMiddleware(
        max_retries=2,
        initial_delay=0.0,
        retry_on=(_RetryableError,),
        on_failure="continue",
    )
    attempts = 0

    def handler(request: ToolCallRequest) -> ToolMessage:
        nonlocal attempts
        attempts += 1
        raise _RetryableError("flaky")

    result = middleware.wrap_tool_call(_make_request(), handler)

    assert attempts == 3
    assert isinstance(result, ToolMessage)
    assert result.status == "error"
    assert "flaky" in result.content
```
