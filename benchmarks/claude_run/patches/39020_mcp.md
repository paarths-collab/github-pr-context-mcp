def _tool_schema_json_for_token_counting(tool: Any) -> str:
    """Serialize a tool to JSON for approximate token counting.

    ``BaseTool`` instances already expose a cached ``tool_call_schema``, so reuse it
    instead of recomputing an equivalent schema through ``convert_to_openai_tool()``.
    The cached schema retains ``$ref``/``$defs``/``title`` fields and is therefore
    slightly larger, which is acceptable for an approximation.

    All other tool types (dicts, callables, ``BaseModel`` subclasses) keep using
    ``convert_to_openai_tool()``.
    """
    # Local imports to avoid circular imports
    from langchain_core.tools import BaseTool
    from langchain_core.utils.function_calling import convert_to_openai_tool

    if isinstance(tool, BaseTool):
        tool_call_schema = tool.tool_call_schema
        parameters = (
            tool_call_schema
            if isinstance(tool_call_schema, dict)
            else tool_call_schema.model_json_schema()
        )
        return json.dumps(
            {
                "type": "function",
                "function": {
                    "name": tool.name,
                    "description": tool.description,
                    "parameters": parameters,
                },
            }
        )

    return json.dumps(convert_to_openai_tool(tool))


def count_tokens_approximately(
    messages: Iterable[MessageLikeRepresentation],
    *,
    chars_per_token: float = 4.0,
    extra_tokens_per_message: float = 3.0,
    count_name: bool = True,
    tools: Sequence[dict[str, Any] | type | Callable | BaseTool] | None = None,
) -> int:
    """Approximate the total number of tokens in messages.

    The token count includes stringified message content, role, and (optionally) name.

    - For AI messages, the token count also includes stringified tool calls.
    - For tool messages, the token count also includes the tool call ID.

    Args:
        messages: List of messages to count tokens for.
        chars_per_token: Number of characters per token to use for the approximation.
            Default is 4 (one token corresponds to ~4 chars for common English text).
            You can also specify float values for more fine-grained control.
            See more here: https://platform.openai.com/tokenizer
        extra_tokens_per_message: Number of extra tokens to add per message.
            Default is 3 (special tokens, including beginning/end of message).
            You can also specify float values for more fine-grained control.
            See more here:
            https://cookbook.openai.com/examples/how_to_count_tokens_with_tiktoken
        count_name: Whether to include message names in the count.
            Enabled by default.
        tools: Optional sequence of tools to include in the token count.

    Returns:
        Approximate number of tokens in the messages.

    Note:
        This is a simple approximation that may not match the exact token count used
        by specific models. For accurate counts, use model-specific tokenizers.

    .. versionadded:: 0.3.46
    """
    token_count = 0.0
    for message in convert_to_messages(messages):
        message_chars = 0
        if isinstance(message.content, str):
            message_chars += len(message.content)

        # TODO: add support for approximate counting for image blocks
        else:
            content = repr(message.content)
            message_chars += len(content)

        if (
            isinstance(message, AIMessage)
            # exclude Anthropic format as tool calls are already included in the content
            and not isinstance(message.content, list)
            and message.tool_calls
        ):
            tool_calls_content = repr(message.tool_calls)
            message_chars += len(tool_calls_content)

        if isinstance(message, ToolMessage):
            message_chars += len(message.tool_call_id)

        role = _get_message_openai_role(message)
        message_chars += len(role)

        if message.name and count_name:
            message_chars += len(message.name)

        # NOTE: we're rounding up per message to ensure that
        # individual message token counts add up to the total count
        # for a list of messages
        token_count += math.ceil(message_chars / chars_per_token)

        # add extra tokens per message
        token_count += extra_tokens_per_message

    if tools:
        for tool in tools:
            tool_chars = len(_tool_schema_json_for_token_counting(tool))
            token_count += math.ceil(tool_chars / chars_per_token)

    # round up once more time in case extra_tokens_per_message is a float
    return math.ceil(token_count)

def test_count_tokens_approximately_base_tool_uses_tool_call_schema() -> None:
    """``BaseTool`` instances should reuse the cached ``tool_call_schema``."""
    from unittest.mock import patch

    from langchain_core.tools import tool

    @tool
    def my_tool(a: int, b: str) -> str:
        """Do something with a and b."""
        return b * a

    messages = [HumanMessage("hi")]
    baseline = count_tokens_approximately(messages)
    with_tool = count_tokens_approximately(messages, tools=[my_tool])
    assert with_tool > baseline

    with patch(
        "langchain_core.utils.function_calling.convert_to_openai_tool"
    ) as mock_convert:
        assert (
            count_tokens_approximately(messages, tools=[my_tool]) == with_tool
        )
        mock_convert.assert_not_called()


def test_count_tokens_approximately_base_tool_with_dict_args_schema() -> None:
    """``tool_call_schema`` may already be a raw dict."""
    from langchain_core.tools import StructuredTool

    my_tool = StructuredTool.from_function(
        func=lambda a: str(a),
        name="my_tool",
        description="Do something with a.",
        args_schema={
            "type": "object",
            "properties": {"a": {"type": "integer"}},
            "required": ["a"],
        },
    )

    messages = [HumanMessage("hi")]
    assert count_tokens_approximately(messages, tools=[my_tool]) > (
        count_tokens_approximately(messages)
    )


def test_count_tokens_approximately_non_base_tool_unchanged() -> None:
    """Dicts, callables and ``BaseModel`` classes keep the existing path."""

    class MyToolSchema(BaseModel):
        """Do something with a and b."""

        a: int
        b: str

    def my_func(a: int, b: str) -> str:
        """Do something with a and b."""
        return b * a

    openai_tool = {
        "type": "function",
        "function": {
            "name": "my_tool",
            "description": "Do something with a and b.",
            "parameters": {
                "type": "object",
                "properties": {"a": {"type": "integer"}, "b": {"type": "string"}},
                "required": ["a", "b"],
            },
        },
    }

    messages = [HumanMessage("hi")]
    baseline = count_tokens_approximately(messages)
    for tool_like in (MyToolSchema, my_func, openai_tool):
        assert count_tokens_approximately(messages, tools=[tool_like]) > baseline
